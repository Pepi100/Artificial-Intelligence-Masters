    # Count diagnosis types from filenames across all folders
    all_diagnoses = []
    for folder in folders:
        folder_path = os.path.join(BASE_PATH, folder)
        for f in os.listdir(folder_path):
            if f.endswith('.hea'):
                filename = f.replace('.hea', '')  # e.g., '21589_hr'
                # Determine diagnosis from suffix
                if '_hr_' in filename or filename == '21589_hr':
                    all_diagnoses.append('healthy')
                elif '_dr_' in filename:
                    all_diagnoses.append('disease')
    
    diagnosis_counts = Counter(all_diagnoses)
    print(f"\n🔍 Diagnosis Distribution from Filenames:")
    for diag, count in sorted(diagnosis_counts.items()):
        print(f"   {diag}: {count:,}")

print("\n✅ Section 3 complete!")

def load_ecg_dat(filepath):
    """Load .dat binary file and return samples"""
    try:
        with open(filepath, 'rb') as f:
            # First ~64 bytes are header metadata, skip it
            data = np.frombuffer(f.read(120000), dtype=np.int16)
        
        if len(data) == 0:
            return None
        
        print(f"  Loaded {len(data):,} samples")
        return data
    except Exception as e:
        print(f"  Error loading: {e}")
        return None

# Find a sample DAT file from any folder

            print(f"\n📊 Loading sample DAT file: {dat_file}")
            
            # Check file size
            file_size = os.path.getsize(dat_file)
            print(f"  File size: {file_size:,} bytes ({file_size/1024:.2f} KB)")
            
            # Load samples (skip first ~64 bytes for header)
            data = load_ecg_dat(dat_file)
            if data is not None:
                # Show sample statistics
                print(f"\n📊 ECG Data Statistics:")
                print(f"  Min value: {data.min():,}")
